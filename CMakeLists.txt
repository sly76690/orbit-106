cmake_minimum_required(VERSION 3.22)
project(Orbit106 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS")
add_subdirectory(vendor/JUCE-8.0.6 EXCLUDE_FROM_ALL)
juce_add_plugin(Orbit106 ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/Source/Assets/orbit-icon.png" COMPANY_NAME "SF Instruments" BUNDLE_ID "com.sfinstruments.orbit106" PLUGIN_MANUFACTURER_CODE SfIn PLUGIN_CODE Orbt FORMATS AU VST3 Standalone PRODUCT_NAME "Orbit 106" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD TRUE VST3_CATEGORIES Instrument Synth)
juce_add_binary_data(OrbitReference SOURCES design/neo-vintage/reference-approved.png)
target_link_libraries(Orbit106 PRIVATE OrbitReference)
target_sources(Orbit106 PRIVATE Source/Plugin.cpp)
if(APPLE)
  target_sources(Orbit106 PRIVATE Source/ForceTouch.mm) # trackpad force on the panel keyboard
endif()
target_compile_definitions(Orbit106 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Orbit106 PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
juce_add_console_app(OrbitSmoke PRODUCT_NAME "Orbit Smoke Tests")
target_sources(OrbitSmoke PRIVATE Tests/Smoke.cpp)
target_link_libraries(OrbitSmoke PRIVATE Orbit106 juce::juce_audio_utils juce::juce_dsp)
target_compile_definitions(OrbitSmoke PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
enable_testing()
add_test(NAME audio_smoke COMMAND OrbitSmoke)

target_compile_definitions(Orbit106 PRIVATE ORBIT_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
